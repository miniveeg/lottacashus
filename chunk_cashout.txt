 1621:       v_balance,

 1627:     return;

 1632:   end if;

 1636:     -- `_gc` alias disambiguates the function-call column from the enclosing

 1639:     select _gc.out_balance into v_balance
 1640:       from public.game_credit(p_user_id, v_payout, v_coin) _gc;

 1642:       set won = true, payout = v_payout, cashed_at = p_cashed_at, completed_at = now()
 1643:       where cb.id = p_bet_id;

 1645:   else
 1646:     update public.crash_bets cb
 1647:       set won = false, payout = 0, completed_at = now()
 1648:       where cb.id = p_bet_id;

 1650:       into v_balance from public.profiles where id = p_user_id;
